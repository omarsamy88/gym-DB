create table member(
member_id number constraint pk_member primary key,
full_name varchar2(100),
email varchar2(100),
phone varchar2(20),
join_date date,
city varchar2 (100)
);

create table membership_plan(
plan_id number constraint pk_membership_plan primary key,
plan_name varchar2(100),
duration_months number,
price number(10,2)
);

create table membership(
membership_id number constraint pk_membership primary key,
member_id number constraint fk_membership_member REFERENCES member(member_id),
plan_id number constraint fk_membership_membership_plan REFERENCES membership_plan(plan_id),
start_date date,
end_date date,
membership_status varchar2(100)
);

create table trainer(
trainer_id number constraint pk_trainer primary key,
full_name varchar2(100),
specialization varchar2(100),
hire_date date
);

create table fitness_class(
class_id number constraint pk_fitness_class primary key,
class_name varchar2(100),
trainer_id number constraint fk_fitness_class_trainer references trainer(trainer_id),
capacity number,
scheduled_at timestamp
);

create table class_booking(
booking_id number constraint pk_class_booking primary key,
member_id number constraint fk_class_booking_member references member(member_id),
class_id number constraint fk_class_booking_fitness_class references fitness_class(class_id),
booking_date date,
attendance_status varchar2(100)
); 

create table payment(
payment_id number constraint pk_payment primary key,
membership_id number constraint fk_payment_membership references membership(membership_id),
payment_date date,
amount number(10,2),
payment_method varchar2(50),
payment_status varchar2(50)
);

select s.membership_id, m.full_name, p.plan_name, start_date, membership_status  from membership s 
join member m on m.member_id=s.member_id
join membership_plan p on p.plan_id=s.plan_id
order by membership_id;

select m.member_id, m.full_name,s.membership_id,s.membership_status from member m 
left join membership s on m.member_id = s.member_id;
---------------------3----------------------------------------------------------------------------
select * from class_booking c 
right join  fitness_class f  on f.class_id = c.class_id;
-----------------------4--------------------------------------------------------------------------
select * from trainer t
left join fitness_class f on t.trainer_id = f.trainer_id;
-----------------------5--------------------------------------------------------------------------
select  p.plan_name, count(s.membership_id) as total_memberships, avg(p.price)as avg_price from membership_plan p 
join membership s on p.plan_id=s.plan_id
group by p.plan_name
order by avg_price desc;

select m.full_name,count(b.member_id) as "number of class"  from member m 
left join class_booking b on m.member_id = b.member_id 
group by full_name
order by "number of class" desc , full_name asc;
---------------------------7----------------------------------------------------------------------------
select  m.plan_name,sum(amount) as "total revenue"  from membership s
join payment p on s.membership_id = p.membership_id
join membership_plan m on s.plan_id = m.plan_id
where lower(p.payment_status) = 'successful'
group by m.plan_name
order by "total revenue" desc;
-----------------------------8------------------------------------------------------------------------
select f.class_id, f.class_name,count(c.class_id) as "number of booking"  from fitness_class f 
left join class_booking c on f.class_id = c.class_id
group by f.class_id, f.class_name
order by "number of booking" desc;

select membership_id,
case 
when membership_status = 'Active' then 'active member'
when membership_status = 'Expired' then 'expired member'
when membership_status = 'pending' then 'pending member'
end AS status_label
from membership
order by membership_id;
---------------------------------10---------------------------------------------------------------------
select full_name, coalesce(phone, email,'no contact details') AS contact_value  from member;
------------------------------11------------------------------------------------------------------------
select s.membership_id,nvl(sum(p.amount), 0) as "total amount"  from membership s
left join payment p on s.membership_id = p.membership_id
group by s.membership_id
order by s.membership_id;
--------------------------------12-------------------------------------------------------------------------
select t.full_name ,count(fc.class_id) as total_classes from trainer t
left join fitness_class fc on t.trainer_id = fc.trainer_id
group by t.trainer_id, t.full_name
order by total_classes desc;
-----------------------------------13---------------------------------------------------------------------
select payment_id, membership_id, payment_method, amount, payment_date,
case
when payment_status = 'Successful' then 'Completed'
when payment_status = 'Pending' then 'Pending'
when payment_status = 'Failed'  then 'Not completed'
end as status_label
from payment
order by payment_date desc;

SELECT payment_method, SUM(amount) AS total_amount,round(AVG(amount),2) AS average_amount, MIN(amount) AS minimum_amount, MAX(amount) AS maximum_amount, COUNT(payment_id) AS total_payments FROM payment
GROUP BY payment_method;
-----------------------------------------------15------------------------------------------------------------
SELECT m.full_name, COUNT(p.payment_id) AS successful_payments, SUM(p.amount) AS total_amount_paid FROM member m
JOIN membership s ON m.member_id = s.member_id
JOIN payment p ON s.membership_id = p.membership_id
WHERE p.payment_status = 'Successful'
GROUP BY m.member_id, m.full_name
HAVING COUNT(p.payment_id) >= 2
ORDER BY total_amount_paid DESC;
